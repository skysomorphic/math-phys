\subsection{Sets}
\lecdate{Lec 1 09-08 Tuesday}

\begin{defn}
    A set is a collection of elements. An element $x$ is in the set $S$, we write $x \in S$. If the element $x$ is not in the set $S$, we write $x \not\in S$.
\end{defn}

\begin{ex}
    Let $P = \{x : x \text{ is a prime number}\}$.
\end{ex}
\begin{defn}
    Two sets $S$ and $T$ are equal if they contain the same elements. In that case we write $S = T$.
\end{defn}

A set can be defined either by listing the elements in the set, or by giving a characterization of these elements.
\begin{gather*}
    A = \set{x : x \text{ is a positive integer dividing }6} \\
    A = \set{1, 2, 3, 6} = \set{2, 6, 1, 3}
\end{gather*}

\begin{defn}
    If $S$ and $T$ are sets such that all elements of $S$ are contained in $T$, we say that $S$ is a subset of $T$ and write $S \subseteq T$. If $S$ is a subset of $T$ and is not equal to $T$, we say that $S$ is a proper subset of $T$ and write $S \subsetneq T$.
\end{defn}

\begin{defn}
    The empty set is the set with no elements, denoted $\varnothing$.
\end{defn}

\begin{defn}
    If $S$ and $T$ are sets, we have two operations
    \begin{itemize}
        \item $S \cup T = \set{x : x \in S \text{ or } T \text{ or both}}$ is called the union of $S$ and $T$.
        \item $S \cap T = \set{x : x \in S \text{ and } T }$ is called the intersection of $S$ and $T$.
    \end{itemize}
\end{defn}
\begin{ex}
    Let $\mathbb{N} = {x \in \mathbb{Z} : x > 0} = \set{1, 2, 3, 4, \dots}$ be the set of natural numbers, and let $-\mathbb{N} = \set{x \in \mathbb{Z} : x < 0}$ be the set of negative integers. Then
    - $\mathbb{N} \cup -\mathbb{N} = {x \in \mathbb{Z} : \not  x= 0}$.
    - $\mathbb{N}\cap \mathbb{N} = \varnothing$
\end{ex}

\begin{defn}
    Let $S_{1}, \dots, S_{k}$ sets. Then, \[
        {}\bigcup^k_{i=1}S_{i} = \{ x: x \in S_{i} \text{ for at least one } 1 \leq i \leq k \}
    \]
    is the union of sets $S_{i}$. Likewise, \[
        {}\bigcap^k_{i=1}S_{i} = \{ x: x \in S_{i}\ \forall\ 1 \leq i \leq k \}
    \]
    is the intersection of sets $S_{i}$.
\end{defn}

\begin{defn}
    Let $S$ and $T$ be sets, The \textbf{Cartesian product} is defined by $S \times T = \{ (s,t): s \in S,\, t \in T\}$.
\end{defn}

\subsection{Functions}

\begin{defn}
    Let $S$ and $T$ be non-empty sets. A function $f$ from $S$ to $T$, denoted $f: S \to T$, is a rule assigning to each $x \in S$ an unique element $f(x) \in T$.
\end{defn}
In that case, we say that
\begin{itemize}
    \item $f(x)$ is the \textbf{image} of $x$ under $f$.
    \item $x$ is the \textbf{preimage} of $f (x)$ under $f$.
    \item $S$ is the \textbf{domain} of $f$.
    \item $T$ is the \textbf{codomain} of $f$.
    \item the \textbf{range} of f is the set ${f (x) : x \in S}$.
    \item if $U \subseteq S$, then the image of $U$ is $f(U) = \{ f(x):x \in S \}$.
    \item if $V \subseteq T$, then the preimage of $V$ is $f^{-1}(V) = \{ x: f(x) \in V \}$.
\end{itemize}

\begin{defn}
    Two functions $f, g : S \to T$ are equal if $f (x) = g(x), \forall x \in S$.
\end{defn}

\begin{defn}
    If every element in the range of $f$ has a unique preimage, we say $f$ is \textbf{injective}, or an \textbf{injection} (one-to-one).
\end{defn}

\begin{defn}
    If every element in the codomain is in the range, we say $f$ is \textbf{surjective}, or a \textbf{surjection} (onto).
\end{defn}

\begin{defn}
    If f is both injective and surjective, we say $f$ is \textbf{bijective}, or a \textbf{bijection}.
\end{defn}
\begin{ex}
    Let $f: \mathbb{Z} \to \mathbb{N}$ be defined by $f(x) = |x| + 1$. Then $f$ is surjective but not injective.
\end{ex}

\begin{defn}
    If $f: S \to T$ and $U \subseteq S$, then the restriction of $f$ to $U$, denoted $f\rvert_{U}$, is defined by $f\rvert_{U}(x) = f (x)$ for all $x \in U$.
\end{defn}

\begin{defn}
    If $f: S \to T$ and $g : T \to U$ are functions, then the function which sends $x \in S$ to $g(f (x))$ is the composite $g \circ f : S \to U$, given $g \circ f (x) = g(f (x))$. Note even if $f, g : S \to S$, we don't necessarily have $f \circ g = g \circ f$.
\end{defn}

\begin{defn}
    A function $f : S \to T$ is \textbf{invertible} if there exists a function $g : T \to S$ such that $g \circ f (x) = x, \forall x \in S$, and $f \circ g(y) = y, \forall y \in T$. If it exists, such a $g$ is the \textbf{inverse} of $f$ and is denoted $f^{-1}$.
\end{defn}

\begin{defn}
    A \textbf{field} $F$ is a set with 2 operations, \textbf{note that these operations are closed}:
    \begin{itemize}
        \item ${}+: F\times F \to F{}$ (\textbf{sum})
        \item ${}\cdot: F\times F \to F{}$ (\textbf{product})
    \end{itemize}
    that obeys the following axioms:
    \begin{itemize}
        \item[(F1)] ${}a+b=b+a{}$ and ${}a \cdot b = b \cdot a{}$ (\textbf{commutativity})
        \item[(F2)] ${}(a+b)+c=a+(b+c){}$ and ${}(a \cdot b) \cdot c = a \cdot (b \cdot c){}$ (\textbf{associativity})
        \item[(F3)] ${}\exists{}$ distinct elements ${}0,1 \in F{}$ s.t. ${}0 + a =a{}$ and ${}1 \cdot a = a{}$ for all ${}a\in F{}$. (\textbf{existence of identities})
        \item[(F4)] ${}\forall a \in F{}$ and ${}\forall b \in F, b \neq 0{}$, there exists ${}c,d \in F{}$ s.t. ${}a+c=0{}$ and ${}b \cdot d = 1{}$. (\textbf{existence of inverses})
        \item[(F5)] ${}\forall a,b,c \in F{}$, then ${}a \cdot (b + c) = a \cdot b + a\cdot c{}$. (\textbf{distributivity of ${}\cdot{}$ over ${}+{}$})
    \end{itemize}
\end{defn}
\begin{ex}
    Some fields:
    \begin{itemize}
        \item $(\RR, +, \cdot)$
        \item $(\QQ, +, \cdot)$
        \item $(\ZZ_{2} = \{ 0,1 \}, +, \cdot)$
    \end{itemize}
\end{ex}

\begin{thm}[Cancellation Result]
    Let ${}F{}$ be a field and ${}a,b,c\in F{}$. Then:
    - if ${}a+c = b+c{}$, then ${}a=b{}$
    - if ${}a \cdot c = b \cdot c{}$, and ${}c \neq 0{}$, then ${}a=b{}$
\end{thm}
\begin{pf}
    Let ${}d \in F{}$ s.t. ${}c \cdot t = 1{}$.
    \begin{align}
        (a \cdot c) \cdot d &= (b \cdot c) \cdot d \\
        a \cdot (c \cdot d) &= b \cdot (c \cdot d) \\
        a \cdot 1  &= b \cdot 1 \\
        a & = b
    \end{align}
    A similar proof exists for the first statement.
\end{pf}
\begin{cor}
    The elements ${}0,1 \in F{}$ are unique. The inverses ${}c,d \in F{}$ are also unique.
\end{cor}

\subsection{Equivalence relations}

\begin{defn}
    Let ${}S{}$ be a set. A relation ${}R{}$ is a subset ${}R \subseteq S \times S{}$. We say that a pair ${}(x,y)\in S{}$ satisfies ${}R{}$ if ${}(x,y) \in R{}$. We denote ${}(x,y) \in R{}$ by ${}x\sim y{}$.
\end{defn}

\begin{defn}
    A relation ${}\sim{}$ on a set ${}S{}$ is a equivalence relation if it satisfies the following properties ${}\forall s,t,u \in S{}$:
    \begin{itemize}
        \item ${}s \sim s{}$ (**reflexivity**)
        \item ${}s \sim t \implies t \sim s{}$ (**symmetry**)
        \item ${}s\sim t{}$ and ${}t \sim u \implies s\sim u{}$ (**transitivity**)
    \end{itemize}
\end{defn}
\begin{exercise} Consider $w \in W$, the set of all english words. What equivalence classes exist on $W$?
\end{exercise}
\begin{ex} Some provided by people:
    \begin{itemize}
        \item ${}w_{1} \sim w_{2}{}$, if they start with the same letter
        \item ${}w_{1} \sim w_{2}{}$, if they start with the same number of characters
        \item ${}w_{1} \sim w_{2}{}$, if they are the same word
            - **for any set you get for free a equivalence relation (equality)!**
            - and also a trivial equivalence relation ${}s \sim_{tr} t, \forall s,t\in S{}$
    \end{itemize}
\end{ex}

\begin{defn}
    If ${}(S, \sim){}$ is a set equipped with a equivalence relation, and ${}t\in S{}$. Let **equivalence class** ${}C_{t} = \{ s\in S: s \sim t \}{}$.
\end{defn}

\begin{thm}
    For ${}(S, \sim){}$, there is a subset ${}T \subseteq S{}$ that satisfies the following:
    \begin{itemize}
        \item ${}\bigcup_{t\in T}{C_{t}} = S {}$
        \item If ${}t,t' \in T{}$ and ${}t\neq t'{}$, then ${}C_{t}\cap C_{t'} = \varnothing{}$.
    \end{itemize}
    ${}T{}$ is a **set of representative** of ${}S{}$ for equivalence relation ${}\sim{}$. Note that this implies ${}C_{t}{}$ partitions ${}S{}$, since they are all pairwise disjoint.
\end{thm}

\begin{thm}[Division with remainder]
    Let ${}m \in \mathbb{N}{}$. For each ${} a \in \mathbb{Z}{}$, there exists a unique ${}r \in \{ 0,1,\dots,m-1 \}{}$ such that ${}a \equiv r \pmod{m}{}$.
\end{thm}
\begin{pf}
    We will show that there exist unique integers ${}k{}$ and ${}r{}$ such that ${}a= km +r{}$ by Lemma~\ref{lm:1}, this give ${}a \equiv r \pmod{m}{}$.

    \textit{Existence.} Consider all the integer multiples of ${}m{}$:\[
        m \mathbb{Z} = \{ 0,\pm m,\pm 2m,\pm 3m,\dots \}.
    \]
    These integers occupy the entire real line by the Archimedean property of the real numbers.

    ![[09-15 Equivalence relations and modular arithmetic-20260915201512028.webp]]

    The integer ${}a{}$ lies somewhere on the real line, in one of these intervals. Thus, it satisfies \[
        km \leq a < (k+1)m
    \]
    for some value ${}k{}$. Subtract ${}km{}$ to obtain \[
        0 \leq a-km < m.
    \]
    Let ${}r=a-km{}$. Then ${}r{}$ satisfies the desired condition.

    \textit{Uniqueness.} Suppose that ${}a=r+km=r'+k'm{}$. Then \[
        r -r = (k'-k)m.
    \]
    It follows that ${}r-r'{}$ is a multiple of ${}m{}$. But since ${}0 \leq r,r' < m{}$, we have ${}-m < r-r' < m{}$. The only multiple of ${}m{}$ on this interval is ${}0{}.$ So ${}r-r' = 0 \implies r=r'{}$ and hence ${}k=k'{}$.
\end{pf}

As a consequence, we can deduce that \[
    \mathbb{Z} = \bigcup_{0}^{m-1} C_{i}
\]
We call ${}\{0, \dots, m-1 \}{}$ the **standard representatives** of their respective **congruence classes** ${}C_{i}{}$ for congruence mod ${}m{}$.

\begin{thm} [Addition and Multiplication of Congruence Classes]
    If ${}a \equiv b \quad (m){}$ and ${}c \equiv d \pmod{m}{}$, then
    $$
    a+c\equiv b+d \pmod{m}, \qquad a \cdot c \equiv b \cdot d \pmod{m}
    $$
\end{thm}

\end{pf}

By the previous theorem, ${}\mathbb{Z} / m\mathbb{Z}{}$ is equipped with addition and multiplication. In practice, we will denote the congruence classes in ${}\mathbb{Z} / m \mathbb{Z}{}$ to be \[

\begin{defn}
    A vector space over a field $F$ is a set $V$ equipped with two (closed) operations,
    \begin{itemize}
        \item $+: V \times V \to V $ (addition) \\ $(x,y) \mapsto x+y$
        \item $\cdot: F \times V$ (scalar multiplication) \\ $(a,x) \mapsto ax$
    \end{itemize}
    and also satisfies the eight following axioms $\forall x,y,z \in V$ and $\forall a,b, \in F$.
    \begin{itemize}
        \item[(VS1)] $x+y=y+x$ \hfill ($+$ is commutative)
        \item[(VS2)] $(x+y)+z = x+(y+z)$ \hfill ($+$ is associative)
        \item[(VS3)] $\exists 0 \in V$ s.t. $x+0=x$ \hfill ($+$ has an identity)
        \item[(VS4)] $\forall x \exists y$ s.t. $x+y=0$ \hfill ($+$ has inverses)
        \item[(VS5)] $1x = x$
        \item[(VS6)] $a(bx) = (ab)x$
        \item[(VS7)] $a(x+y) = ax + ay$
        \item[(VS8)] $(a+b)x = ax + bx$
    \end{itemize}
\end{defn}
\begin{remark}
    Note that (VS1-4) is equivalent to saying that $(V, +)$ forms an abelian group, and that $F$ is only relevant for the latter four axioms. Also, note that $0 \in V$ and $0 \in F$ is not necessarily exactly the same compounds.
\end{remark}
\begin{remark}
    A vector space \textit{need not} have multiplication defined on vectors!
\end{remark}
\begin{ex} Some vector spaces include:
    \begin{itemize}
        \item Every field is a vector space over itself. (pretty trivial)
        \item $F^n = \{ (a_{1},a_{2},\dots,a_{n}) \mid a_{i} \in F \}$ for any field $F$ is a vector space over $F$.
        \item $\CC = \{ a+bi \mid a,b \in \RR \}$ is a vector space over $\RR$, and also a vector space over $\QQ$!
        \item Let $k \in  \ZZ_{>0}$. $P_{k}(F) = \{ a_{0} + a_{1}x + \dots + a_{k}x^k \mid a,b \in F \}$, the set of polynomials with degree $k$ and coefficients from $F$, is a vector space over $F$.
        \item $P(f) = \bigcup_{k \in \ZZ_{>0}} P_{k}(F)$ is a vector space of all polynomials over $F$.
        \item Let $S$ be a nonempty set, and a $F$ be a field. $\mathcal{F}(S,F)$ is the set of all functions from $S \to F$. $\mathcal{F}$ is a vector space over $F$, with addition and scalar multiplication defined as follows, for $f, g \in \mathcal{F}$, $s \in S$, $a \in F$.
            \begin{gather*}
                (f+g)(s) = f(s) \\
                (af)(s) = a \cdot f(s)
            \end{gather*}
    \end{itemize}
\end{ex}

\begin{thm}[Simple properties of vector spaces]
    Let $V$ be a vector space over $F$, $v,x,y,z \in V$, and $a \in F$. Then

        \item If $a \in F$, then $(-a)b = a(-v) = -(av)$.
    \end{enumerate}
\end{thm}

\subsection{Subspaces of vector spaces}
\begin{defn}
    Let $V$ be a vector space over $F$. $W \subseteq V$ is a \textbf{subspace} of $V$ if it is a vector space equipped with the restriction of operations of $V$.
\end{defn}

\begin{thm}[VERY IMPORTANT AND USEFUL]\label{thm:subspace}
    Let $V$ be a vector space over $F$. Then $W \subseteq V$ is a subspace if and only if it satisfies:
    \begin{enumerate}[label=(\roman*)]
        \item $W \neq \varnothing$
        \item $W$ is closed under addition
        \item $W$ is closed under scalar multiplication
    \end{enumerate}
\end{thm}
\begin{pf}
    SEE PDF
\end{pf}
\begin{ex}
    Consider the vector space $\RR^3 = \{ (a,b,c) \mid a,b,c \in \RR \}$ over $\RR$.
    \begin{itemize}
        \item $\RR^3$ is a subspace of itself. (trivial)
        \item $W_{1} = \{ (a,b,c) \mid c=0 \}$ is a subspace. ($\cong\RR^2$)
        \item $W_{2} = \{ (a,b,c) \mid 71a-22b+7c = 0 \}$ is a subspace. ($\cong\RR$)
        \item $W_{3} = \{ (a,b,c) \mid a+b=-c^{2} \}$ is NOT a subspace. ($v = (-2,-2,-2) \in W_{3}$ but $2v = (-4,-4,-4) \not\in W_{3}$.)
    \end{itemize}
\end{ex}

\begin{thm}
    Let $V$ be a vector space and $W_{1}$ and $W_{2}$ be subspaces of $V$. Then, $W_{1} \cap W_{2}$ is also a subspace.
\end{thm}
\begin{pf}
    We will prove the conditions in Theorem~\ref{thm:subspace}
    \begin{enumerate}[label=(\roman*)]
        \item $0 \in W_{1}$ and $0 \in W_{2}$, so $0 \in W_{1} \cap W_{2}$.
        \item $x, y \in W_{1} \cap W_{2} \implies x+y \in W_{1}$ and $x+y \in W_{2}$ because addition is closed in subspaces, so $x+y \in W_{1} \cap W_{2}$.
        \item $x \in W_{1} \cap W_{2} \implies cx \in W_{1}$ and $cx \in W_{2}$ because scalar multiplication is closed in subspaces, so $cx \in W_{1} \cap W_{2}$. \qedhere
    \end{enumerate}
\end{pf}

\begin{defn}
    Let $V$ be a vector space, and $S$ be nonempty subset of $V$. We say $v \in V$ is a \textbf{linear combination} of $s \in S$ if there exists $u_{1}, u_{2}, \dots, u_{n} \in S$ and $a_{1}, a_{2}, \dots, a_{n} \in F$ s.t. \[
        v = a_{1}u_{1} + a_{2}u_{2} + \dots + a_{n}u_{n}
    \]
    We call $a_{i}$ \textbf{coefficients}.
\end{defn}
\begin{ex}\label{ex:0lincomb}
    $0$ is always a linear combination of any set of vectors.
\end{ex}
\begin{ex}
    Consider the vector space of polynomials $P(\QQ)$ over $\QQ$. $3x^{2} + 1$ is a linear combination of $x^{2}$ and $1$, or even $-7x^{2} + 1$ and $4x^{2} + 2$. (The reader may check the latter as an exercise.)
\end{ex}

\begin{defn}
    Let $S$ be a nonempty subset of vector space $V$. The \textbf{span} of $S$ is the set of linear combinations of $S$, denoted $\opspan(S)$. By definition, $\opspan(\varnothing) =  \{ 0 \}$.
\end{defn}

\begin{thm}
    Let $S$ be a subset of a vector space $V$. $\opspan(S) = W$ is a subspace of $V$. Also, $S \subseteq W$. Moreover, any subspace of $V$ that contains $S$, also contains $\opspan(S)$
\end{thm}
\begin{pf}
    Using Theorem~\ref{thm:subspace} we show that $\opspan(S)$ is a subspace by the following:
    \begin{enumerate}[label=(\roman*)]
        \item $0 \in \opspan(S)$ (see Example~\ref{ex:0lincomb}) even if $S$ is empty, since by definition $\opspan(\varnothing) = \{ 0 \}$. So, $\opspan(S)$ is not empty.
        \item $\opspan(S)$ is closed under addition. too lazy to write this out, its intuitive.
        \item $\opspan(S)$ is closed under scalar multiplication. also too lazy lmao, same logic, just write out the definitions.
    \end{enumerate}
    For every $s \in S$, $s$ can be written as a linear combination of itself by $1 \cdot s = s$, so it is in $\opspan(S) = W$, so $S \subseteq W$.

    Finally, we show that every subspace of $V$ that contains $S$ also contains $\opspan(S)$. By definition, subspaces are closed under addition and scalar multiplication, which are exactly the operations that make up linear combinations. So, $\opspan(S)$ must also be in that subspace, as it is exactly the linear combinations of $S$.
\end{pf}
